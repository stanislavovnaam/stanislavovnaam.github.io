Весёлое окно с мороженым на Python

Инструкция для детей 8–12 лет: код спрятан по блокам и открывается по ссылке.

1. Подготовка

  1. Установите Python 3.
  2. Создайте папку, например icecream_game.
  3. Скачайте картинку мороженого и назовите её icecream.png.
  4. Положите картинку в эту же папку.
  5. Создайте файл icecream_game.py.

2. Шаг 1 — подключаем модули

Сначала объясните детям, что модули — это помощники для программы.

Показать код шага 1 
import tkinter as tk
import random
import ctypes

3. Шаг 2 — список фраз и счётчик

Здесь мы сохраняем шутки, которые окно будет говорить. Можно поменять их на свои.

Показать код шага 2 
NO_TEXTS = [
    "Ну пожалуйста...",
    "Подумай ещё раз 😏",
    "Мороженое уже почти ждёт!",
    "Это твой последний шанс 🍦",
    "Ну хотя бы одно маленькое!"
]

current_text = 0

4. Шаг 3 — закрытие и скрытие окна

Этот шаг делает окно «упрямым»: крестик не работает, а из панели задач его не видно.

Показать код функции on_close 
def on_close():
    pass
Показать код функции hide_from_taskbar 
def hide_from_taskbar(win):
    try:
        hwnd = ctypes.windll.user32.GetParent(win.winfo_id())
        GWL_EXSTYLE = -20
        WS_EX_TOOLWINDOW = 0x00000080
        WS_EX_APPWINDOW = 0x00040000

        style = ctypes.windll.user32.GetWindowLongW(hwnd, GWL_EXSTYLE)
        style = style | WS_EX_TOOLWINDOW
        style = style & ~WS_EX_APPWINDOW
        ctypes.windll.user32.SetWindowLongW(hwnd, GWL_EXSTYLE, style)

        win.withdraw()
        win.after(10, win.deiconify)
    except:
        pass

5. Шаг 4 — движение окна и кнопка «Нет»

Здесь окно будет прыгать по экрану, но не вылетать за его края.

Показать код функции move_popup 
def move_popup():
    popup.update_idletasks()

    screen_w = popup.winfo_screenwidth()
    screen_h = popup.winfo_screenheight()

    win_w = 520
    win_h = 360

    max_x = max(0, screen_w - win_w)
    max_y = max(0, screen_h - win_h)

    x = random.randint(0, max_x)
    y = random.randint(0, max_y)

    popup.geometry(f"{win_w}x{win_h}+{x}+{y}")
Показать код функции say_no 
def say_no():
    global current_text
    current_text = (current_text + 1) % len(NO_TEXTS)
    label.config(text=NO_TEXTS[current_text])
    move_popup()

6. Шаг 5 — победа и закрытие

Если нажать «Хорошо, куплю», программа покажет сообщение о победе и через секунду закроется.

Показать код функции say_yes 
def say_yes():
    label.config(
        text="Победа! 🍦\nМороженое будет куплено.\nОкна сейчас закроются."
    )
    popup.after(800, close_all)
Показать код функции close_all 
def close_all():
    popup.destroy()
    root.destroy()

7. Шаг 6 — создаём окно с кнопками

Это самый большой блок. Он создаёт всплывающее окно, картинку, надписи и кнопки.

Показать код функции open_popup 
def open_popup():
    global popup, label, ice_image, current_text
    current_text = 0

    popup = tk.Toplevel(root)
    popup.title("Мороженое и кибер-шутка")
    popup.geometry("520x360")
    popup.resizable(False, False)
    popup.configure(bg="#fff4d6")
    popup.protocol("WM_DELETE_WINDOW", on_close)
    popup.overrideredirect(True)

    top_frame = tk.Frame(popup, bg="#fff4d6")
    top_frame.pack(pady=15)

    try:
        ice_image = tk.PhotoImage(file="icecream.png")
        img_label = tk.Label(top_frame, image=ice_image, bg="#fff4d6")
        img_label.pack()
    except:
        tk.Label(top_frame, text="🍦", font=("Arial", 40), bg="#fff4d6").pack()

    title = tk.Label(
        popup,
        text="Пока не купишь мороженое,\nокно будет висеть 😈",
        font=("Arial", 18, "bold"),
        fg="#c62828",
        bg="#fff4d6"
    )
    title.pack(pady=(5, 10))

    label = tk.Label(
        popup,
        text=NO_TEXTS[0],
        font=("Arial", 14),
        fg="#5d4037",
        bg="#fff4d6"
    )
    label.pack(pady=5)

    button_frame = tk.Frame(popup, bg="#fff4d6")
    button_frame.pack(pady=20)

    btn_yes = tk.Button(
        button_frame,
        text="Хорошо, куплю",
        width=18,
        height=2,
        command=say_yes,
        bg="#66bb6a",
        fg="white",
        activebackground="#43a047",
        activeforeground="white",
        font=("Arial", 12, "bold"),
        relief="flat"
    )
    btn_yes.grid(row=0, column=0, padx=10)

    btn_no = tk.Button(
        button_frame,
        text="Нет, не куплю",
        width=18,
        height=2,
        command=say_no,
        bg="#ef5350",
        fg="white",
        activebackground="#d32f2f",
        activeforeground="white",
        font=("Arial", 12, "bold"),
        relief="flat"
    )
    btn_no.grid(row=0, column=1, padx=10)

    popup.transient(root)
    popup.grab_set()
    popup.focus_set()

    root.after(50, hide_from_taskbar, popup)
    root.wait_window(popup)

8. Шаг 7 — главное окно

Последний кусочек — само главное окно, которое открывает шутку.

Показать код главного окна 
root = tk.Tk()
root.title("Главное окно")
root.geometry("340x180")
root.configure(bg="#e3f2fd")
root.overrideredirect(True)

main_label = tk.Label(
    root,
    text="Нажми кнопку, чтобы открыть шутку",
    font=("Arial", 13, "bold"),
    bg="#e3f2fd",
    fg="#1565c0"
)
main_label.pack(pady=(30, 15))

open_button = tk.Button(
    root,
    text="Показать шутку",
    command=open_popup,
    width=22,
    height=2,
    bg="#42a5f5",
    fg="white",
    activebackground="#1e88e5",
    activeforeground="white",
    font=("Arial", 12, "bold"),
    relief="flat"
)
open_button.pack()

root.mainloop()

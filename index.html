# antivirus_v1.py
import os
import datetime

VIRUS_SIGNATURES = ['malware_test_1', 'badcode_xyz', 'eicar_test_string']
LOG_FILE = "antivirus_v1_log.txt"

def log_message(msg):
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(f"[{datetime.datetime.now()}] {msg}\n")

def scan_file(path):
    try:
        with open(path, 'r', errors='ignore') as f:
            content = f.read()
        for sig in VIRUS_SIGNATURES:
            if sig in content:
                return True
    except Exception as e:
        log_message(f"Ошибка чтения файла {path}: {e}")
    return False

def scan_folder(folder):
    found = []
    total = 0
    for root, _, files in os.walk(folder):
        for name in files:
            total += 1
            p = os.path.join(root, name)
            if scan_file(p):
                found.append(p)
                log_message(f"Подозрительный файл: {p}")
    log_message(f"Проверено файлов: {total}, найдено: {len(found)}")
    return found, total

def show_menu():
    print("\n=== Простой антивирус (v1) ===")
    print("1) Сканировать папку")
    print("2) Выход")

def main():
    while True:
        show_menu()
        choice = input("Выбор: ").strip()
        if choice == "1":
            folder = input("Папка для проверки: ").strip() or "."
            infected, total = scan_folder(folder)
            print(f"\nПроверено файлов: {total}")
            if infected:
                print("⚠ Найдены подозрительные файлы:")
                for f in infected:
                    print(" -", f)
            else:
                print("✅ Чисто!")
        elif choice == "2":
            break
        else:
            print("Неверный выбор, попробуйте снова.")

if __name__ == "__main__":
    main()

# antivirus_v2.py
import os
import datetime

SIGNATURES = ['virus_alpha', 'trojan_sample', 'test_mal_99']
REPORT_FILE = "antivirus_v2_report.txt"

COLOR_RED = "\033[91m"
COLOR_GREEN = "\033[92m"
COLOR_YELLOW = "\033[93m"
COLOR_RESET = "\033[0m"

def has_virus(filepath):
    try:
        data = open(filepath, 'r', errors='ignore').read()
        return any(s in data for s in SIGNATURES)
    except:
        return False

def check_directory(dirpath):
    bad = []
    total = 0
    for root, _, files in os.walk(dirpath):
        for fname in files:
            total += 1
            full = os.path.join(root, fname)
            if has_virus(full):
                bad.append(full)
    return bad, total

def write_report(bad, total):
    with open(REPORT_FILE, "w", encoding="utf-8") as f:
        f.write("Отчёт антивируса v2\n")
        f.write(f"Время: {datetime.datetime.now()}\n")
        f.write(f"Проверено файлов: {total}\n")
        f.write(f"Найдено подозрительных: {len(bad)}\n")
        if bad:
            f.write("Файлы:\n")
            for x in bad:
                f.write(x + "\n")

def show_menu():
    print(COLOR_YELLOW + "\n=== Цветной антивирус (v2) ===" + COLOR_RESET)
    print("1) Сканировать папку")
    print("2) Выход")

    while True:

        choice = input("Выбор: ").strip()
        if choice == "1":
            d = input("Введите путь: ").strip() or "."
            res, total = check_directory(d)
            write_report(res, total)
            print(f"\nПроверено файлов: {total}")
            if res:
                print(COLOR_RED + "!!! Обнаружены файлы:" + COLOR_RESET)
                for x in res:
                    print(" >", x)
                print(COLOR_GREEN + f"Отчёт сохранён в {REPORT_FILE}" + COLOR_RESET)
            else:
                print(COLOR_GREEN + "Всё чисто." + COLOR_RESET)
        elif choice == "2":

# antivirus_v3.py
import os, hashlib

KNOWN_BAD_HASHES = {
    'd41d8cd98f00b204e9800998ecf8427e',
    '5d41402abc4b2a76b9719d911017c592',
}
TARGET_EXTS = {'.txt', '.py', '.bat', '.cmd', '.exe'}

def file_md5(path):
    h = hashlib.md5()
    try:
        with open(path, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b''):
                h.update(chunk)
        return h.hexdigest()
    except:
        return None

def scan_dir(root):
    infected = []
    total = 0
    for r, _, fs in os.walk(root):
        for name in fs:
            _, ext = os.path.splitext(name)
            if ext.lower() not in TARGET_EXTS:
                continue
            total += 1
            p = os.path.join(r, name)
            h = file_md5(p)
            if h and h in KNOWN_BAD_HASHES:
                infected.append(p)
    return infected, total

def show_menu():
    print("\n=== MD5‑антивирус (v3) ===")
    print("1) Сканировать папку")
    print("2) Выход")

def main():
    while True:
        show_menu()
        choice = input("Выбор: ").strip()
        if choice == "1":
            folder = input("Сканировать папку: ").strip() or "."
            bad, total = scan_dir(folder)
            print(f"\nПроверено файлов (по расширениям): {total}")
            print("🔍 Результат:", "чисто" if not bad else f"найдено {len(bad)}")
            for f in bad:
                print(" -", f)
        elif choice == "2":
            break
        else:
            print("Неверный выбор, попробуйте снова.")

if __name__ == "__main__":
    main()

# antivirus_v9.py
import os

SIGNATURES = ['low_v9', 'mid_v9', 'high_v9']

def risk_level(path):
    try:
        txt = open(path, 'r', errors='ignore').read()
        count = sum(1 for s in SIGNATURES if s in txt)
        return count
    except:
        return 0

def scan(folder):
    res = []
    total = 0
    for r, _, fs in os.walk(folder):
        for name in fs:
            total += 1
            p = os.path.join(r, name)
            lvl = risk_level(p)
            if lvl > 0:
                res.append((p, lvl))
    return res, total

def show_menu():
    print("\n=== Антивирус с уровнем опасности v9 ===")
    print("1) Сканировать папку")
    print("2) Выход")

    while True:
        show_menu()

        if choice == "1":
            folder = input("Папка: ").strip() or "."
            data, total = scan(folder)
            print(f"Проверено файлов: {total}")
            if not data:
                print("✅ Всё чисто")
            else:
                print("🔎 Результаты:")
                for path, lvl in data:
                    print(f"• {path} — уровень: {lvl}")
        elif choice == "2":
            break
        else:
            print("Неверный выбор, попробуйте снова.")

# antivirus_v10.py
import os, shutil

SIGNATURES = ['quarantine_v10', 'bad_10', 'test_10']
QUARANTINE_DIR = "quarantine_v10"

def is_bad(path):
    try:
        t = open(path, 'r', errors='ignore').read()
        return any(s in t for s in SIGNATURES)
    except:
        return False

def quarantine_move(src):
    os.makedirs(QUARANTINE_DIR, exist_ok=True)
    name = os.path.basename(src)
    dst = os.path.join(QUARANTINE_DIR, name)
    shutil.move(src, dst)
    return dst

def scan_and_quarantine(folder):
    moved = []
    total = 0
    for root, _, files in os.walk(folder):
        for name in files:
            total += 1
            p = os.path.join(root, name)
            if is_bad(p):
                new_path = quarantine_move(p)
                moved.append((p, new_path))
    return moved, total

def show_menu():
    print("\n=== Антивирус с карантином v10 ===")
    print("1) Сканировать и переместить в карантин")
    print("2) Выход")

def main():
    while True:
        show_menu()
        choice = input("Выбор: ").strip()
        if choice == "1":
            folder = input("Папка: ").strip() or "."
            res, total = scan_and_quarantine(folder)
            print(f"Всего файлов: {total}")
            print("Перемещено в карантин:", len(res))
            for old, new in res:
                print(old, "->", new)
        elif choice == "2":
            break
        else:
            print("Неверный выбор, попробуйте снова.")

if __name__ == "__main__":
    main()

# antivirus_v1.py
import os
import datetime
import tkinter as tk
from tkinter import messagebox, filedialog

VIRUS_SIGNATURES = ['malware_test_1', 'badcode_xyz', 'eicar_test_string']
LOG_FILE = "antivirus_v1_log.txt"

def log_message(msg):
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(f"[{datetime.datetime.now()}] {msg}\n")

def scan_file(path):
    try:
        with open(path, 'r', errors='ignore') as f:
            content = f.read()
        for sig in VIRUS_SIGNATURES:
            if sig in content:
                return True
    except Exception as e:
        log_message(f"Ошибка чтения файла {path}: {e}")
    return False

def scan_folder(folder):
    found = []
    total = 0
    for root, _, files in os.walk(folder):
        for name in files:
            total += 1
            p = os.path.join(root, name)
            if scan_file(p):
                found.append(p)
                log_message(f"Подозрительный файл: {p}")
    log_message(f"Проверено файлов: {total}, найдено: {len(found)}")
    return found, total

def start_scan():
    folder = filedialog.askdirectory(title="Выберите папку")
    if not folder:
        return
    status_label.config(text="🔍 Сканирование...", fg="blue")
    root.update()
    infected, total = scan_folder(folder)
    status_label.config(text=f"✅ Проверено: {total}", fg="green")
    if infected:
        messagebox.showwarning("⚠ Внимание", f"Найдено подозрительных файлов: {len(infected)}")
        result_text.delete(1.0, tk.END)
        for f in infected:
            result_text.insert(tk.END, f + "\n")
    else:
        messagebox.showinfo("✅ Готово", "Всё чисто!")

# Создаём окно
root = tk.Tk()
root.title("🛡 Антивирус v1")
root.iconbitmap("icon.ico")
root.geometry("600x400")

title_label = tk.Label(root, text="🛡 Простой антивирус", font=("Arial", 16, "bold"))
title_label.pack(pady=10)

scan_btn = tk.Button(root, text="📁 Сканировать папку", command=start_scan, font=("Arial", 12))
scan_btn.pack(pady=10)

# antivirus_v3.py
import os, hashlib
import tkinter as tk
from tkinter import messagebox, filedialog

KNOWN_BAD_HASHES = {
    'd41d8cd98f00b204e9800998ecf8427e',
    '5d41402abc4b2a76b9719d911017c592',
}
TARGET_EXTS = {'.txt', '.py', '.bat', '.cmd', '.exe'}

def file_md5(path):
    h = hashlib.md5()
    try:
        with open(path, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b''):
                h.update(chunk)
        return h.hexdigest()
    except:
        return None

def scan_dir(root_path):
    infected = []
    total = 0
    for r, _, fs in os.walk(root_path):
        for name in fs:
            _, ext = os.path.splitext(name)
            if ext.lower() not in TARGET_EXTS:
                continue
            total += 1
            p = os.path.join(r, name)
            h = file_md5(p)
            if h and h in KNOWN_BAD_HASHES:
                infected.append(p)
    return infected, total

def start_scan():
    folder = filedialog.askdirectory(title="Выберите папку")
    if not folder:
        return
    status_label.config(text="🔍 Сканирование...", fg="blue")

        result_text.delete(1.0, tk.END)

    try:
        txt = open(path, 'r', errors='ignore').read()
        count = sum(1 for s in SIGNATURES if s in txt)
        return count
    except:
        return 0

    res = []

    for r, _, fs in os.walk(folder):

# antivirus_v10.py
import os, shutil
import tkinter as tk
from tkinter import messagebox, filedialog

SIGNATURES = ['quarantine_v10', 'bad_10', 'test_10']
QUARANTINE_DIR = "quarantine_v10"

def is_bad(path):
    try:
        t = open(path, 'r', errors='ignore').read()
        return any(s in t for s in SIGNATURES)
    except:
        return False

def quarantine_move(src):
    os.makedirs(QUARANTINE_DIR, exist_ok=True)
    name = os.path.basename(src)
    dst = os.path.join(QUARANTINE_DIR, name)
    shutil.move(src, dst)
    return dst

def scan_and_quarantine(folder):
    moved = []
    total = 0
    for root, _, files in os.walk(folder):
        for name in files:
            total += 1
            p = os.path.join(root, name)
            if is_bad(p):
                new_path = quarantine_move(p)
                moved.append((p, new_path))
    return moved, total

def start_scan():
    folder = filedialog.askdirectory(title="Выберите папку")
    if not folder:
        return
    status_label.config(text="🔍 Сканирование...", fg="blue")
    root_win.update()
    res, total = scan_and_quarantine(folder)
    status_label.config(text=f"✅ Проверено: {total}", fg="green")
    messagebox.showinfo("Готово", f"В карантин: {len(res)}")

# antivirus_v14.py
import os
import tkinter as tk
from tkinter import messagebox, filedialog

MAX_SIZE_MB = 10  # максимальный нормальный размер
MIN_SIZE_BYTES = 0  # минимальный размер

def check_size_anomaly(path):
    try:
        size = os.path.getsize(path)
        size_mb = size / (1024 * 1024)
        if size_mb > MAX_SIZE_MB or size < MIN_SIZE_BYTES:
            return True, size
        return False, size
    except:
        return False, 0

def scan_folder(folder):
    anomalies = []
    total = 0
    for root, _, files in os.walk(folder):
        for name in files:
            total += 1
            p = os.path.join(root, name)
            is_anomaly, size = check_size_anomaly(p)
            if is_anomaly:
                anomalies.append((p, size))
    return anomalies, total

def start_scan():

    if not folder:
        return

            size_kb = size / 1024

# antivirus_v15.py
import os
import datetime
import tkinter as tk
from tkinter import messagebox, filedialog

SUSPICIOUS_WORDS = [
    'password', 'secret', 'admin', 'hack', 'crack',
    'virus', 'malware', 'trojan', 'keylog', 'backdoor'
]
REPORT_FILE = "antivirus_v15_report.txt"

def check_keywords(path):
    found_words = []
    try:
        with open(path, 'r', errors='ignore') as f:
            content = f.read().lower()
        for word in SUSPICIOUS_WORDS:
            if word in content:
                found_words.append(word)
        return found_words
    except:
        return []

def scan_folder(folder):
    results = []
    total = 0
    for root, _, files in os.walk(folder):
        for name in files:
            total += 1
            p = os.path.join(root, name)
            words = check_keywords(p)
            if words:
                results.append((p, words))
    return results, total

def write_report(results, total):
    with open(REPORT_FILE, "w", encoding="utf-8") as f:
        f.write("Отчёт антивируса v15\n")
        f.write(f"Время: {datetime.datetime.now()}\n")
        f.write(f"Проверено файлов: {total}\n")
        f.write(f"Найдено подозрительных: {len(results)}\n\n")
        for path, words in results:
            f.write(f"Файл: {path}\n")
            f.write(f"Слова: {', '.join(words)}\n\n")

def start_scan():
    folder = filedialog.askdirectory(title="Выберите папку")

        return
    status_label.config(text="🔍 Сканирование...", fg="blue")

    else:
        messagebox.showinfo("✅ Готово", "Подозрительных слов не найдено!")
    result_text.insert(tk.END, f"\nОтчёт: {REPORT_FILE}\n")

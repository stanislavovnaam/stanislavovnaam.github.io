Весёлое окно с мороженым на Python

Инструкция для детей 8–12 лет и их родителей/учителей.

1. Идея программы

1.1 Что делает наша программа

  • Показывает окно с картинкой мороженого.
  • Пишет шутливый текст про мороженое.
  • Кнопка «Нет, не куплю» заставляет окно прыгать по экрану.
  • Кнопка «Хорошо, куплю» показывает сообщение о победе и закрывает программу.

1.2 Для чего это нужно

  • Чтобы потренироваться писать программы с окнами.
  • Чтобы увидеть, как кнопки вызывают разные функции.
  • Чтобы поиграть с друзьями и родителями.

2. Подготовка перед кодом

2.1 Устанавливаем инструменты

  1. Установите Python 3 (лучше версия 3.10 или новее).
  2. Проверьте, что команда python работает в командной строке.

2.2 Готовим папку с файлами

  1. Создайте папку, например icecream_game.
  2. Скачайте картинку мороженого и назовите её icecream.png.
  3. Положите картинку в эту же папку.
  4. Создайте файл icecream_game.py в этой папке.

3. Код программы

3.1 Общий вид

Ниже — весь код программы. Его можно просто скопировать в файл icecream_game.py. Комментарии (строки с #) помогают понять, что происходит.

3.2 Полный код

# Импортируем нужные модули
# tkinter — чтобы рисовать окна и кнопки
# random — чтобы окно прыгало по случайным местам
# ctypes — нужен, чтобы спрятать окно из панели задач Windows
import tkinter as tk
import random
import ctypes

# Это разные шутливые фразы,
# которые будут меняться, когда ребёнок нажимает «Нет, не куплю»
NO_TEXTS = [
    "Ну пожалуйста...",
    "Подумай ещё раз 😏",
    "Мороженое уже почти ждёт!",
    "Это твой последний шанс 🍦",
    "Ну хотя бы одно маленькое!"
]

current_text = 0  # номер текущей фразы

def on_close():
    # Функция вызывается, если пользователь пытается нажать крестик.
    # Мы ничего не делаем, чтобы окно нельзя было закрыть крестиком.
    pass

def hide_from_taskbar(win):
    """
    Эта функция прячет окно из панели задач Windows.
    Детям можно сказать: «мы делаем окно невидимым внизу экрана».
    """
    try:
        hwnd = ctypes.windll.user32.GetParent(win.winfo_id())
        GWL_EXSTYLE = -20
        WS_EX_TOOLWINDOW = 0x00000080
        WS_EX_APPWINDOW = 0x00040000

        style = ctypes.windll.user32.GetWindowLongW(hwnd, GWL_EXSTYLE)
        style = style | WS_EX_TOOLWINDOW
        style = style & ~WS_EX_APPWINDOW
        ctypes.windll.user32.SetWindowLongW(hwnd, GWL_EXSTYLE, style)

        # Коротко прячем окно и показываем снова, чтобы стиль применился
        win.withdraw()
        win.after(10, win.deiconify)
    except:
        # Если что-то пошло не так — просто ничего не делаем
        pass

def move_popup():
    """Эта функция двигает всплывающее окно в случайное место на экране.
    Мы следим, чтобы окно не выходило за край.
    """
    popup.update_idletasks()

    screen_w = popup.winfo_screenwidth()   # ширина экрана
    screen_h = popup.winfo_screenheight()  # высота экрана

    win_w = 520   # ширина окна
    win_h = 360   # высота окна

    # Максимальные координаты, чтобы окно не вылезло за экран
    max_x = max(0, screen_w - win_w)
    max_y = max(0, screen_h - win_h)

    # Случайные координаты внутри экрана
    x = random.randint(0, max_x)
    y = random.randint(0, max_y)

    # Перемещаем окно
    popup.geometry(f"{win_w}x{win_h}+{x}+{y}")

def say_no():
    """Что происходит, когда ребёнок нажимает кнопку «Нет, не куплю».
    Меняем текст на следующий и двигаем окно.
    """
    global current_text
    current_text = (current_text + 1) % len(NO_TEXTS)
    label.config(text=NO_TEXTS[current_text])
    move_popup()

def say_yes():
    """Что происходит, когда ребёнок нажимает «Хорошо, куплю».
    Показываем сообщение о победе и через небольшой таймер
    закрываем все окна.
    """
    label.config(
        text="Победа! 🍦\nМороженое будет куплено.\nОкна сейчас закроются."
    )
    popup.after(800, close_all)  # через 0.8 секунды вызываем закрытие

def close_all():
    """Закрываем всплывающее окно и главное окно."""
    popup.destroy()
    root.destroy()

def open_popup():
    """Создаём всплывающее окно с мороженым и кнопками."""
    global popup, label, ice_image, current_text
    current_text = 0  # начинаем с первой фразы

    # Создаём новое окно поверх главного
    popup = tk.Toplevel(root)
    popup.title("Мороженое и кибер-шутка")
    popup.geometry("520x360")
    popup.resizable(False, False)
    popup.configure(bg="#fff4d6")              # светлый фон
    popup.protocol("WM_DELETE_WINDOW", on_close)
    popup.overrideredirect(True)              # убираем рамку и заголовок

    # Верхний блок для картинки
    top_frame = tk.Frame(popup, bg="#fff4d6")
    top_frame.pack(pady=15)

    # Пытаемся загрузить файл icecream.png.
    # Если файла нет, показываем эмодзи.
    try:
        ice_image = tk.PhotoImage(file="icecream.png")
        img_label = tk.Label(top_frame, image=ice_image, bg="#fff4d6")
        img_label.pack()
    except:
        tk.Label(top_frame, text="🍦", font=("Arial", 40), bg="#fff4d6").pack()

    # Заголовок окна
    title = tk.Label(
        popup,
        text="Пока не купишь мороженое,\nокно будет висеть 😈",
        font=("Arial", 18, "bold"),
        fg="#c62828",
        bg="#fff4d6"
    )
    title.pack(pady=(5, 10))

    # Надпись, которая будет меняться при «Нет, не куплю»
    label = tk.Label(
        popup,
        text=NO_TEXTS[0],
        font=("Arial", 14),
        fg="#5d4037",
        bg="#fff4d6"
    )
    label.pack(pady=5)

    # Блок для двух кнопок
    button_frame = tk.Frame(popup, bg="#fff4d6")
    button_frame.pack(pady=20)

    # Кнопка «Да»
    btn_yes = tk.Button(
        button_frame,
        text="Хорошо, куплю",
        width=18,
        height=2,
        command=say_yes,
        bg="#66bb6a",
        fg="white",
        activebackground="#43a047",
        activeforeground="white",
        font=("Arial", 12, "bold"),
        relief="flat"
    )
    btn_yes.grid(row=0, column=0, padx=10)

    # Кнопка «Нет»
    btn_no = tk.Button(
        button_frame,
        text="Нет, не куплю",
        width=18,
        height=2,
        command=say_no,
        bg="#ef5350",
        fg="white",
        activebackground="#d32f2f",
        activeforeground="white",
        font=("Arial", 12, "bold"),
        relief="flat"
    )
    btn_no.grid(row=0, column=1, padx=10)

    popup.transient(root)
    popup.grab_set()
    popup.focus_set()

    # Прячем всплывающее окно из панели задач
    root.after(50, hide_from_taskbar, popup)

    # Ждём, пока всплывающее окно не закроется
    root.wait_window(popup)

# Создаём главное окно программы
root = tk.Tk()
root.title("Главное окно")
root.geometry("340x180")
root.configure(bg="#e3f2fd")
root.overrideredirect(True)  # убираем рамку для главного окна

# Надпись в главном окне
main_label = tk.Label(
    root,
    text="Нажми кнопку, чтобы открыть шутку",
    font=("Arial", 13, "bold"),
    bg="#e3f2fd",
    fg="#1565c0"
)
main_label.pack(pady=(30, 15))

# Кнопка, которая открывает всплывающее окно
open_button = tk.Button(
    root,
    text="Показать шутку",
    command=open_popup,
    width=22,
    height=2,
    bg="#42a5f5",
    fg="white",
    activebackground="#1e88e5",
    activeforeground="white",
    font=("Arial", 12, "bold"),
    relief="flat"
)
open_button.pack()

# Запускаем главный цикл программы
root.mainloop()

4. Как запустить программу

4. Шаги запуска

  1. Сохраните файл icecream_game.py и картинку icecream.png в одной папке.
  2. Откройте командную строку (Terminal, PowerShell или cmd).
  3. Перейдите в папку с файлом командой cd.
  4. Запустите программу командой: python icecream_game.py.

5. Задания для детей

  • Добавьте ещё 2–3 свои смешные фразы в список NO_TEXTS.
  • Поменяйте текст победы в функции say_yes() на свой.
  • Измените цвета кнопок (параметры bg и fg).
